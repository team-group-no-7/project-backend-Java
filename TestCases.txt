   📡 1. REST API SPECIFICATIONS & cURL COMMANDS 

  ### 1️⃣ GET Learner Profile
  • Endpoint: GET /api/profile/{userId}
  • Purpose: Fetch learner profile info.
  • cURL:
    curl -X GET http://localhost:8080/api/profile/1 \
      -H "Authorization: Bearer <your_jwt_token>"
    
  • Response (200 OK):
    {
      "id": 1,
      "name": "Riya Raj",
      "email": "riya@learnhub.com",
      "role": "LEARNER",
      "avatarUrl": "https://example.com/avatar.jpg",
      "headline": "Full Stack Java Learner",
      "location": "Bangalore, India"
    }
    
  ──────
  ### 2️⃣ PUT Update Learner Profile

  • Endpoint: PUT /api/profile/{userId}
  • Purpose: Update learner profile details.
  • cURL:
    curl -X PUT http://localhost:8080/api/profile/1 \
      -H "Content-Type: application/json" \
      -H "Authorization: Bearer <your_jwt_token>" \
      -d '{
        "name": "Riya Raj",
        "headline": "Senior Java Backend Engineer",
        "location": "Mumbai, India",
        "avatarUrl": "https://example.com/new_avatar.jpg"

   🧪 2. UNIT TEST CASES SUITE (JUnit 5 + Mockito) 

  Below are the unit tests located in your src/test/java directory:

  ### 📄 1. Profile Service Tests (ProfileServiceTest.java)

    @ExtendWith(MockitoExtension.class)
    class ProfileServiceTest {
    
        @Mock private UserRepository userRepository;
        @InjectMocks private ProfileServiceImpl profileService;
    
        @Test
        void getProfile_Success() {
            User user = new User();
            user.setId(1L); user.setName("Riya Raj"); user.setEmail("riya@learnhub.com");
    
            when(userRepository.findById(1L)).thenReturn(Optional.of(user));
    
            ProfileResponse response = profileService.getProfile(1L);
            assertNotNull(response);
            assertEquals("Riya Raj", response.getName());
        }
    
        @Test
        void getProfile_NotFound() {
            when(userRepository.findById(99L)).thenReturn(Optional.empty());
            assertThrows(ResourceNotFoundException.class, () -> profileService.getProfile(99L));
        }
    }
    
  ### 📄 2. Dashboard Service Tests (DashboardServiceTest.java)

    @ExtendWith(MockitoExtension.class)
    class DashboardServiceTest {
    
        @Mock private PurchaseRepository purchaseRepository;
        @InjectMocks private DashboardServiceImpl dashboardService;
    
        @Test
        void getDashboard_Success() {
            when(purchaseRepository.findLibraryByUserId(1L)).thenReturn(List.of(new Purchase()));
            when(purchaseRepository.totalInvestment(1L)).thenReturn(new BigDecimal("499.00"));
    
            DashboardResponse response = dashboardService.getDashboard(1L);
            assertEquals(1L, response.getActiveResources());
            assertEquals(new BigDecimal("499.00"), response.getTotalInvestment());
        }
    
        @Test
        void getDashboard_ZeroInvestment() {
            when(purchaseRepository.findLibraryByUserId(1L)).thenReturn(List.of());
            when(purchaseRepository.totalInvestment(1L)).thenReturn(null);
    
            DashboardResponse response = dashboardService.getDashboard(1L);
            assertEquals(0L, response.getActiveResources());
            assertEquals(BigDecimal.ZERO, response.getTotalInvestment());
        }
    }
    
  ### 📄 3. Purchase Service Tests (PurchaseServiceTest.java)

    @ExtendWith(MockitoExtension.class)
    class PurchaseServiceTest {

        @Mock private PurchaseRepository purchaseRepository;
        @InjectMocks private PurchaseServiceImpl purchaseService;

        @Test
        void getPurchaseHistory_Success() {
            when(purchaseRepository.findByUserIdOrderByPurchasedAtDesc(1L)).thenReturn(List.of(new Purchase()));

            List<PurchaseResponse> history = purchaseService.getPurchaseHistory(1L);
            assertNotNull(history);
            verify(purchaseRepository, times(1)).findByUserIdOrderByPurchasedAtDesc(1L);
        }
    }
